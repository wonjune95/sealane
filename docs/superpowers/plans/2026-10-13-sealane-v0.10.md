**Goal:** 로컬 계정을 그대로 둔 채 OIDC(코드 플로우 + PKCE, 그룹→역할 매핑) 로그인을 추가하고, mkdocs-material 문서 사이트를 GitHub Pages에 올려 `v0.10.0`을 발행한다.

**Architecture:** OIDC는 `internal/auth/oidc.go`(go-oidc + x/oauth2, 지연 discovery)와 `internal/api/oidc.go`(세 핸들러)로 들어간다. state·nonce·PKCE verifier는 기존 `Sessions`의 HMAC으로 서명한 10분짜리 쿠키 `sealane_oidc`에 담고, 성공하면 기존 `sealane_session` 쿠키를 발급하므로 무상태 HA가 유지된다. 세션 페이로드에 `provider`(local|oidc)가 추가되어 UI가 SSO 사용자에게 비밀번호 변경을 숨긴다. 실패는 HTML 없이 `/?oidc_error=<code>`로 돌려보내고 로그인 화면이 메시지를 보여준다. 차트는 `oidc.*` 값을 env로 주입한다. 문서 사이트는 README 내용을 페이지로 쪼개 mkdocs-material로 빌드하고 Actions가 Pages에 배포한다.

## Global Constraints

- 의존성 추가는 `github.com/coreos/go-oidc/v3`(v3.21.0) 하나와 `golang.org/x/oauth2`의 직접 의존 전환뿐. 테스트의 JWT 서명은 표준 라이브러리로 직접 만든다(go-jose를 직접 의존으로 올리지 않는다). npm 의존성 추가 없음.
- 로컬 로그인은 항상 유지. OIDC는 `OIDC_ISSUER_URL`이 비어 있으면 완전히 꺼진다.
- 환경변수(차트 값): `OIDC_ISSUER_URL`(`oidc.issuerURL`), `OIDC_CLIENT_ID`(`oidc.clientID`), `OIDC_CLIENT_SECRET`(`oidc.clientSecretRef{name,key}`→secretKeyRef), `OIDC_REDIRECT_URL`(`oidc.redirectURL`, 명시 필수), `OIDC_SCOPES`(`oidc.scopes`, 기본 `openid profile email groups`, 공백 구분), `OIDC_GROUPS_CLAIM`(`oidc.groupsClaim`, 기본 `groups`), `OIDC_USERNAME_CLAIM`(`oidc.usernameClaim`, 기본 `email`), `OIDC_ROLES_ADMIN`/`_EDITOR`/`_VIEWER`(`oidc.roles.admin|editor|viewer`, 쉼표 구분 그룹), `OIDC_DEFAULT_ROLE`(`oidc.defaultRole`, 기본 `viewer`, 빈값 = 매핑 없는 사용자 거부), `OIDC_DISPLAY_NAME`(`oidc.displayName`, 기본 `SSO`). 여러 그룹에 속하면 admin > editor > viewer.
- 라우트: `GET /api/auth/methods`(비인증), `GET /api/auth/oidc/login`, `GET /api/auth/oidc/callback`. 상태 쿠키 `sealane_oidc`(HttpOnly, SameSite=Lax, Path `/api/auth/oidc`, 10분). 세션 쿠키 `sealane_session` 페이로드 `user|role|exp|provider`(3필드짜리 옛 세션은 무효 → 0.10 배포 시 한 번 재로그인).
- 실패 리다이렉트: `/?oidc_error=state`(쿠키 없음·state 불일치·만료), `/?oidc_error=access_denied`(그룹 매핑 없음), `/?oidc_error=idp`(IdP 에러 파라미터, discovery/교환/검증 실패). 응답 본문에 IdP 에러 원문을 싣지 않고 `slog.Warn`으로만 남긴다.

- 차트 version/appVersion `0.10.0`. 기존 values 키 유지. `oidc.enabled=true`인데 `issuerURL`/`clientID`/`clientSecretRef.name`/`redirectURL` 중 하나라도 비면 `fail`. nil-safe 관용구(`default dict`) 유지.

type Identity struct {
	Name     string
	Role     string // admin | editor | viewer
	Provider string // ProviderLocal | ProviderOIDC
}

func (s *Sessions) Sign(id Identity, ttl time.Duration) string {
	provider := id.Provider
	if provider == "" {
		provider = ProviderLocal
	}
	payload := fmt.Sprintf("%s|%s|%d|%s",
		base64.RawURLEncoding.EncodeToString([]byte(id.Name)), id.Role, time.Now().Add(ttl).Unix(), provider)
	return payload + "|" + s.sig(payload)
}

var ErrSessionInvalid = errors.New("login required")

func (s *Sessions) Parse(value string) (Identity, error) {
	parts, err := s.verify(value, 4)
	if err != nil {
		return Identity{}, ErrSessionInvalid
	}
	exp, err := strconv.ParseInt(parts[2], 10, 64)
	if err != nil || time.Now().Unix() > exp {
		return Identity{}, ErrSessionInvalid
	}
	name, err := base64.RawURLEncoding.DecodeString(parts[0])
	if err != nil || !validRoles[parts[1]] || (parts[3] != ProviderLocal && parts[3] != ProviderOIDC) {
		return Identity{}, ErrSessionInvalid
	}
	return Identity{Name: string(name), Role: parts[1], Provider: parts[3]}, nil
}

// SignState protects the OIDC login state (state, nonce, PKCE verifier) in a short-lived
// cookie with the same HMAC key, so the flow stays stateless across replicas.
func (s *Sessions) SignState(state, nonce, verifier string, ttl time.Duration) string {
	payload := fmt.Sprintf("%s|%s|%s|%d", state, nonce, verifier, time.Now().Add(ttl).Unix())
	return payload + "|" + s.sig(payload)
}

var ErrStateInvalid = errors.New("login state invalid or expired")

func (s *Sessions) ParseState(value string) (state, nonce, verifier string, err error) {
	parts, err := s.verify(value, 4)
	if err != nil {
		return "", "", "", ErrStateInvalid
	}
	exp, err := strconv.ParseInt(parts[3], 10, 64)
	if err != nil || time.Now().Unix() > exp {
		return "", "", "", ErrStateInvalid
	}
	return parts[0], parts[1], parts[2], nil
}

// verify checks the trailing HMAC and splits the payload into exactly n fields.
func (s *Sessions) verify(value string, n int) ([]string, error) {
	i := strings.LastIndexByte(value, '|')
	if i < 0 {
		return nil, ErrSessionInvalid
	}
	payload, sig := value[:i], value[i+1:]
	if !hmac.Equal([]byte(s.sig(payload)), []byte(sig)) {
		return nil, ErrSessionInvalid
	}
	parts := strings.Split(payload, "|")
	if len(parts) != n {
		return nil, ErrSessionInvalid
	}
	return parts, nil
}

// MapRole picks the highest app role whose group list intersects the user's groups;
// defaultRole applies when nothing matches ("" = refuse the login).
func MapRole(groups []string, roles map[string][]string, defaultRole string) (string, bool) {
	has := map[string]bool{}
	for _, g := range groups {
		has[g] = true
	}
	for _, role := range []string{"admin", "editor", "viewer"} {
		for _, g := range roles[role] {
			if has[g] {
				return role, true
			}
		}
	}
	if defaultRole != "" && validRoles[defaultRole] {
		return defaultRole, true
	}
	return "", false
}

func signRS256(t *testing.T, key *rsa.PrivateKey, claims map[string]any) string {
	t.Helper()
	b64 := base64.RawURLEncoding.EncodeToString
	body, _ := json.Marshal(claims)
	signing := b64([]byte(`{"alg":"RS256","typ":"JWT","kid":"k1"}`)) + "." + b64(body)
	h := sha256.Sum256([]byte(signing))
	sig, err := rsa.SignPKCS1v15(rand.Reader, key, crypto.SHA256, h[:])
	if err != nil {
		t.Fatal(err)
	}
	return signing + "." + b64(sig)
}

	ClientSecret  string
	RedirectURL   string

func OIDCConfigFromEnv() (*OIDCConfig, error) {

	if issuer == "" {
		return nil, nil
	}
	c := &OIDCConfig{
		IssuerURL:     issuer,
		ClientID:      os.Getenv("OIDC_CLIENT_ID"),

		UsernameClaim: envOr("OIDC_USERNAME_CLAIM", "email"),
		Roles: map[string][]string{
			"admin":  splitList(os.Getenv("OIDC_ROLES_ADMIN"), ","),
			"editor": splitList(os.Getenv("OIDC_ROLES_EDITOR"), ","),
			"viewer": splitList(os.Getenv("OIDC_ROLES_VIEWER"), ","),
		},
		DefaultRole: os.Getenv("OIDC_DEFAULT_ROLE"),
		DisplayName: envOr("OIDC_DISPLAY_NAME", "SSO"),

		return nil, fmt.Errorf("OIDC_DEFAULT_ROLE %q is not one of admin, editor, viewer", c.DefaultRole)

func envOr(key, def string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return def
}

func splitList(s, sep string) []string {
	var out []string
	for _, p := range strings.Split(s, sep) {
		if p = strings.TrimSpace(p); p != "" {
			out = append(out, p)
		}
	}
	return out

// ErrNoRole means the IdP authenticated the user but no group maps to a Sealane role.
var ErrNoRole = errors.New("account has no role in Sealane")

// OIDC runs the authorization-code + PKCE login against one provider. Discovery is lazy
// so an unreachable IdP never blocks the boot or local logins.
type OIDC struct {
	cfg      OIDCConfig
	mu       sync.Mutex
	provider *oidc.Provider
}

func NewOIDC(cfg OIDCConfig) *OIDC { return &OIDC{cfg: cfg} }

func (o *OIDC) DisplayName() string { return o.cfg.DisplayName }

func (o *OIDC) discover(ctx context.Context) (*oidc.Provider, error) {
	o.mu.Lock()
	defer o.mu.Unlock()
	if o.provider != nil {
		return o.provider, nil
	}
	ctx, cancel := context.WithTimeout(ctx, 10*time.Second)
	defer cancel()
	p, err := oidc.NewProvider(ctx, o.cfg.IssuerURL)
	if err != nil {
		return nil, fmt.Errorf("oidc discovery: %w", err)
	}
	o.provider = p
	return p, nil
}

func (o *OIDC) oauth(p *oidc.Provider) oauth2.Config {
	return oauth2.Config{
		ClientID: o.cfg.ClientID, ClientSecret: o.cfg.ClientSecret, RedirectURL: o.cfg.RedirectURL,
		Endpoint: p.Endpoint(), Scopes: o.cfg.Scopes,
	}
}

// AuthCodeURL is the IdP redirect; verifier is the PKCE secret the caller keeps in the state cookie.
func (o *OIDC) AuthCodeURL(ctx context.Context, state, nonce, verifier string) (string, error) {
	p, err := o.discover(ctx)
	if err != nil {
		return "", err
	}
	return o.oauth(p).AuthCodeURL(state, oidc.Nonce(nonce), oauth2.S256ChallengeOption(verifier)), nil
}

// Exchange redeems the callback code, verifies the id_token (signature, audience, expiry,
// nonce) and maps the user's groups to a Sealane role.
func (o *OIDC) Exchange(ctx context.Context, code, verifier, nonce string) (Identity, error) {
	p, err := o.discover(ctx)
	if err != nil {
		return Identity{}, err
	}
	tok, err := o.oauth(p).Exchange(ctx, code, oauth2.VerifierOption(verifier))
	if err != nil {
		return Identity{}, fmt.Errorf("token exchange: %w", err)
	}
	raw, _ := tok.Extra("id_token").(string)
	if raw == "" {
		return Identity{}, errors.New("token response has no id_token")
	}
	idt, err := p.Verifier(&oidc.Config{ClientID: o.cfg.ClientID}).Verify(ctx, raw)
	if err != nil {
		return Identity{}, fmt.Errorf("id_token: %w", err)
	}
	if idt.Nonce != nonce {
		return Identity{}, errors.New("id_token nonce mismatch")
	}
	var claims map[string]any
	if err := idt.Claims(&claims); err != nil {
		return Identity{}, err
	}
	name, _ := claims[o.cfg.UsernameClaim].(string)
	if name == "" {
		name = idt.Subject
	}
	var groups []string
	if gs, ok := claims[o.cfg.GroupsClaim].([]any); ok {
		for _, g := range gs {
			if s, ok := g.(string); ok {
				groups = append(groups, s)
			}
		}
	}
	role, ok := MapRole(groups, o.cfg.Roles, o.cfg.DefaultRole)
	if !ok {
		return Identity{}, ErrNoRole
	}
	return Identity{Name: name, Role: role, Provider: ProviderOIDC}, nil
}

const (
	oidcStateCookie = "sealane_oidc"
	oidcStatePath   = "/api/auth/oidc"
	oidcStateTTL    = 10 * time.Minute
)

// SetOIDC enables single sign-on; nil (the default) leaves only local accounts.
func (s *Server) SetOIDC(o *auth.OIDC) { s.oidc = o }

// handleAuthMethods tells the login page which methods exist. Unauthenticated by design.
func (s *Server) handleAuthMethods(w http.ResponseWriter, _ *http.Request) {
	out := map[string]any{"local": !s.dev}
	if s.oidc != nil {
		out["oidc"] = map[string]any{"enabled": true, "displayName": s.oidc.DisplayName()}
	}
	writeJSON(w, http.StatusOK, out)
}

func randomToken() string {
	b := make([]byte, 16)
	if _, err := rand.Read(b); err != nil {
		panic(err) // no entropy — nothing sane to do
	}
	return base64.RawURLEncoding.EncodeToString(b)
}

// handleOIDCLogin starts the authorization-code flow: state, nonce and the PKCE verifier
// travel in a signed cookie so any replica can finish the flow.
func (s *Server) handleOIDCLogin(w http.ResponseWriter, r *http.Request) {
	if s.oidc == nil {
		writeError(w, http.StatusNotFound, "NotFound", "single sign-on is not configured")
		return
	}
	state, nonce, verifier := randomToken(), randomToken(), oauth2.GenerateVerifier()
	target, err := s.oidc.AuthCodeURL(r.Context(), state, nonce, verifier)
	if err != nil {
		slog.Warn("oidc login", "err", err)
		http.Redirect(w, r, "/?oidc_error=idp", http.StatusFound)
		return
	}
	http.SetCookie(w, &http.Cookie{
		Name: oidcStateCookie, Value: s.sessions.SignState(state, nonce, verifier, oidcStateTTL),
		Path: oidcStatePath, HttpOnly: true, SameSite: http.SameSiteLaxMode, MaxAge: int(oidcStateTTL.Seconds()),
	})
	http.Redirect(w, r, target, http.StatusFound)
}

// handleOIDCCallback finishes the flow. Failures never leak IdP details to the browser:
// they are logged and the login page shows a generic message keyed by oidc_error.
func (s *Server) handleOIDCCallback(w http.ResponseWriter, r *http.Request) {
	fail := func(code string, err error) {
		slog.Warn("oidc callback failed", "reason", code, "err", err)
		clearCookie(w, oidcStateCookie, oidcStatePath)
		http.Redirect(w, r, "/?oidc_error="+code, http.StatusFound)
	}
	if s.oidc == nil {
		fail("idp", errors.New("single sign-on is not configured"))
		return
	}
	q := r.URL.Query()
	if e := q.Get("error"); e != "" {
		fail("idp", errors.New(e+": "+q.Get("error_description")))
		return
	}
	c, err := r.Cookie(oidcStateCookie)
	if err != nil {
		fail("state", err)
		return
	}
	state, nonce, verifier, err := s.sessions.ParseState(c.Value)
	if err != nil || state == "" || state != q.Get("state") {
		fail("state", errors.New("state mismatch or expired"))
		return
	}
	id, err := s.oidc.Exchange(r.Context(), q.Get("code"), verifier, nonce)
	if errors.Is(err, auth.ErrNoRole) {
		fail("access_denied", err)
		return
	}
	if err != nil {
		fail("idp", err)
		return
	}
	clearCookie(w, oidcStateCookie, oidcStatePath)
	setSessionCookie(w, s.sessions.Sign(id, sessionTTL))
	slog.Info("login", "user", id.Name, "role", id.Role, "provider", id.Provider)
	http.Redirect(w, r, "/", http.StatusFound)
}

```yaml
name: docs

on:
  push:
    branches: [main]
    paths: [docs/**, mkdocs.yml, .github/workflows/docs.yml]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@<sha> # v4
      - uses: actions/setup-python@<sha> # v5
        with:
          python-version: "3.12"
      - run: pip install -r docs/requirements.txt
      - run: mkdocs build --strict
      - uses: actions/configure-pages@<sha> # v5
        with:
          enablement: true
      - uses: actions/upload-pages-artifact@<sha> # v4
        with:
          path: site
  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@<sha> # v4
```
(`pin actions/setup-python v5; pin actions/configure-pages v5; pin actions/upload-pages-artifact v4; pin actions/deploy-pages v4`; checkout SHA는 ci.yml의 것.) `.gitignore`에 `site/` 추가.
