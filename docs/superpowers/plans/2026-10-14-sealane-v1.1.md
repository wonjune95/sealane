```go
type ref struct {
	TypeID    string `json:"typeId"`
	Kind      string `json:"kind"`
	Namespace string `json:"namespace"`
	Name      string `json:"name"`
}

type finding struct {
	Severity string `json:"severity"` // "error" | "warning"
	Code     string `json:"code"`
	Message  string `json:"message"`
	Resource ref    `json:"resource"`
	Related  []ref  `json:"related,omitempty"`
	Service  string `json:"service,omitempty"` // "ns/name" of the Service the finding concerns
}

type analysis struct {
	Findings []finding `json:"findings"`
	Skipped  []string  `json:"skipped,omitempty"` // checks that could not run: "pods" (no RBAC)
	Checked  int       `json:"checked"`           // routing resources inspected
}

// analyzeMesh inspects the whole cluster and returns findings whose Resource
// lives in ns (ns == "" → all). Cross-namespace references are always resolved
// cluster-wide; only the reporting scope is filtered.
func analyzeMesh(ctx context.Context, c *k8s.Clients, ns string) analysis
```

Finding codes (exact strings; the UI and e2e depend on them):

| Code | Severity | On | When |
|---|---|---|---|
| `vs-host-missing` | error | VirtualService | a route destination resolves to an in-cluster Service that does not exist |
| `vs-subset-undefined` | error | VirtualService | a destination names a subset no DestinationRule defines for that Service |
| `vs-gateway-missing` | error | VirtualService | `spec.gateways` names an Istio Gateway that does not exist (`mesh` ignored) |
| `vs-host-conflict` | warning | VirtualService | another VirtualService bound to the same gateway claims the same host (`related` = the others) |
| `backend-no-endpoints` | warning | VirtualService / HTTPRoute | the backend Service exists but has zero ready endpoints |
| `httproute-backend-missing` | error | HTTPRoute | a `backendRefs` Service does not exist |
| `httproute-not-accepted` | error | HTTPRoute | any `status.parents[].conditions` with type `Accepted` or `ResolvedRefs` and status `False` (message = `Reason: Message`) |
| `gateway-not-programmed` | error | Gateway (API) | `status.conditions` `Accepted` or `Programmed` is `False` |
| `dr-host-missing` | warning | DestinationRule | `spec.host` resolves in-cluster but the Service is missing and no ServiceEntry covers the host |
| `dr-subset-no-pods` | warning | DestinationRule | Service selector + subset labels match no Pod in the Service's namespace (needs pods) |
| `gateway-selector-no-pods` | warning | Gateway (Istio) | `spec.selector` matches no Pod in any namespace (needs pods) |

When listing Pods returns Forbidden, the two pod checks are skipped and `Skipped` contains `"pods"`. Any other list error for a kind (CRD not installed) skips that kind silently, like the flow map.

```go
func vs(ns, name string, hosts, gateways []string, dests ...*istioapi.Destination) *istionet.VirtualService {
	var routes []*istioapi.HTTPRouteDestination

		routes = append(routes, &istioapi.HTTPRouteDestination{Destination: d})

	return &istionet.VirtualService{
		ObjectMeta: metav1.ObjectMeta{Namespace: ns, Name: name},
		Spec:       istioapi.VirtualService{Hosts: hosts, Gateways: gateways, Http: []*istioapi.HTTPRoute{{Route: routes}}},
	}
}

func codes(a analysis) map[string]int {
	m := map[string]int{}
	for _, f := range a.Findings {
		m[f.Code]++
	}
	return m
```

```go
package api

import (
	"context"
	"fmt"
	"net/http"
	"sort"
	"strings"

	istioapi "istio.io/api/networking/v1"
	corev1 "k8s.io/api/core/v1"
	apierrors "k8s.io/apimachinery/pkg/api/errors"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/labels"
	gwapi "sigs.k8s.io/gateway-api/apis/v1"

	"github.com/wonjune95/sealane/internal/k8s"
)

// (types ref, finding, analysis as in Interfaces)

const (
	typeVS       = "virtualservices.networking.istio.io"
	typeDR       = "destinationrules.networking.istio.io"
	typeIstioGW  = "gateways.networking.istio.io"
	typeHTTPRoute = "httproutes.gateway.networking.k8s.io"
	typeGateway  = "gateways.gateway.networking.k8s.io"
)

// meshState is everything analyzeMesh reads, fetched once. Lists that fail
// (CRD not installed, RBAC) are nil and their checks are skipped.
type meshState struct {
	services map[string]*corev1.Service // ns/name
	pods     map[string][]labels.Set    // ns → pod label sets; nil when pods are not readable
	podsOK   bool
	ses      []flowServiceEntry
	drs      []subsetDef // one row per subset (DR with no subsets still produces one row with Subset "")
	drLabels map[string]map[string]string // "ns/dr/subset" → subset labels
	istioGWs map[string]*istioapi.Gateway // ns/name
	epCache  map[string]int
}

func (s *Server) handleAnalyze(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, analyzeMesh(r.Context(), clientFrom(r.Context()), r.URL.Query().Get("ns")))
}

func analyzeMesh(ctx context.Context, c *k8s.Clients, ns string) analysis {
	out := analysis{Findings: []finding{}}
	st := loadMeshState(ctx, c)
	if !st.podsOK {
		out.Skipped = append(out.Skipped, "pods")
	}
	inScope := func(resNS string) bool { return ns == "" || resNS == ns }
	add := func(f finding) {
		if inScope(f.Resource.Namespace) {
			out.Findings = append(out.Findings, f)
		}
	}
	svcKey := func(b flowBackend) string { return b.Namespace + "/" + b.Name }

	// --- VirtualServices ---
	type gwHost struct{ gw, host string }
	claims := map[gwHost][]ref{}
	if vss, err := c.Istio.NetworkingV1().VirtualServices("").List(ctx, metav1.ListOptions{}); err == nil {
		for i := range vss.Items {
			v := vss.Items[i]
			self := ref{TypeID: typeVS, Kind: "VirtualService", Namespace: v.Namespace, Name: v.Name}
			out.Checked++
			for _, g := range v.Spec.Gateways {
				if g == "mesh" {
					continue
				}
				full := g
				if !strings.Contains(g, "/") {
					full = v.Namespace + "/" + g
				}
				if _, ok := st.istioGWs[full]; !ok && st.istioGWs != nil {
					add(finding{Severity: "error", Code: "vs-gateway-missing", Resource: self,
						Message: fmt.Sprintf("gateway %q does not exist", full)})
				}
				for _, h := range v.Spec.Hosts {
					claims[gwHost{full, h}] = append(claims[gwHost{full, h}], self)
				}
			}
			for _, d := range vsDestinations(&v.Spec) {
				b := resolveDestination(d.Host, v.Namespace, 0)
				if b.External {
					continue
				}
				svc, ok := st.services[svcKey(b)]
				if !ok {
					add(finding{Severity: "error", Code: "vs-host-missing", Resource: self, Service: svcKey(b),
						Message: fmt.Sprintf("destination %q is not a Service (looked for %s)", d.Host, svcKey(b))})
					continue
				}
				if st.readyEndpoints(ctx, c, svc.Namespace, svc.Name) == 0 {
					add(finding{Severity: "warning", Code: "backend-no-endpoints", Resource: self, Service: svcKey(b),
						Message: fmt.Sprintf("Service %s has no ready endpoints", svcKey(b))})
				}
				if d.Subset != "" && !st.subsetDefined(b, d.Subset) {
					add(finding{Severity: "error", Code: "vs-subset-undefined", Resource: self, Service: svcKey(b),
						Message: fmt.Sprintf("subset %q is not defined in any DestinationRule for %s", d.Subset, svcKey(b))})
				}
			}
		}
	}
	for k, refs := range claims {
		if len(refs) < 2 {
			continue
		}
		for i, r := range refs {
			others := append(append([]ref{}, refs[:i]...), refs[i+1:]...)
			add(finding{Severity: "warning", Code: "vs-host-conflict", Resource: r, Related: others,
				Message: fmt.Sprintf("host %q on gateway %s is also claimed by %d other VirtualService(s); rule order between them is undefined", k.host, k.gw, len(others))})
		}
	}

	// --- DestinationRules ---
	// (one finding per DR for a missing host; one per subset without pods)
	// --- Istio Gateways: selector → pods ---
	// --- HTTPRoutes: status, backends ---
	// --- Gateway API Gateways: status ---

	sort.SliceStable(out.Findings, func(i, j int) bool {
		a, b := out.Findings[i], out.Findings[j]
		if a.Severity != b.Severity {
			return a.Severity == "error"
		}
		ka, kb := a.Resource.Namespace+"/"+a.Resource.Name, b.Resource.Namespace+"/"+b.Resource.Name
		if ka != kb {
			return ka < kb
		}
		return a.Code < b.Code
	})
	return out
}
```

```go
// vsDestinations flattens http/tls/tcp route destinations.
func vsDestinations(spec *istioapi.VirtualService) []*istioapi.Destination {
	var out []*istioapi.Destination
	for _, h := range spec.Http {
		for _, rt := range h.Route {
			if rt.Destination != nil {
				out = append(out, rt.Destination)
			}
		}
	}
	for _, t := range spec.Tls {
		for _, rt := range t.Route { if rt.Destination != nil { out = append(out, rt.Destination) } }
	}
	for _, t := range spec.Tcp {
		for _, rt := range t.Route { if rt.Destination != nil { out = append(out, rt.Destination) } }

func loadMeshState(ctx context.Context, c *k8s.Clients) *meshState {
	st := &meshState{services: map[string]*corev1.Service{}, epCache: map[string]int{}, drLabels: map[string]map[string]string{}}
	if svcs, err := c.Kube.CoreV1().Services("").List(ctx, metav1.ListOptions{}); err == nil {
		for i := range svcs.Items {
			s := &svcs.Items[i]
			st.services[s.Namespace+"/"+s.Name] = s
		}
	}
	if pods, err := c.Kube.CoreV1().Pods("").List(ctx, metav1.ListOptions{}); err == nil {
		st.podsOK = true
		st.pods = map[string][]labels.Set{}
		for i := range pods.Items {
			p := &pods.Items[i]
			st.pods[p.Namespace] = append(st.pods[p.Namespace], labels.Set(p.Labels))
		}
	} else if !apierrors.IsForbidden(err) {
		st.podsOK = true // transient error: do not advertise a permission problem; checks simply find no pods
		st.pods = map[string][]labels.Set{}
		// ponytail: a transient list failure reads as "no pods"; acceptable for a 30 s refresh page
	}
	// ServiceEntries, DestinationRules (+ drLabels), Istio Gateways: same shape as handleFlowMap / handleRouteMatch.
	return st
}

// anyPodMatches reports whether a label selector matches a pod in ns ("" = any namespace).
func (st *meshState) anyPodMatches(ns string, sel map[string]string) bool {
	s := labels.SelectorFromSet(labels.Set(sel))
	check := func(sets []labels.Set) bool {
		for _, ls := range sets {
			if s.Matches(ls) {
				return true
			}
		}
		return false
	}
	if ns != "" {
		return check(st.pods[ns])
	}
	for _, sets := range st.pods {
		if check(sets) {
			return true
		}
	}
	return false
}
```

```go
// statusOf condenses a resource's status into "", "ok", "error:<Reason>" or
// "warning:<Reason>" for the list pill. Gateway API: parents[].conditions /
// conditions; Istio: status.validationMessages (present when istiod analysis is on).
func statusOf(typeID string, u *unstructured.Unstructured) string {
	switch {
	case strings.HasSuffix(typeID, ".gateway.networking.k8s.io"):
		var conds []any
		if typeID == "gateways.gateway.networking.k8s.io" || typeID == "gatewayclasses.gateway.networking.k8s.io" {
			conds, _, _ = unstructured.NestedSlice(u.Object, "status", "conditions")
		} else {
			parents, _, _ := unstructured.NestedSlice(u.Object, "status", "parents")
			for _, p := range parents {
				if pm, ok := p.(map[string]any); ok {
					c, _, _ := unstructured.NestedSlice(pm, "conditions")
					conds = append(conds, c...)
				}
			}
		}
		if len(conds) == 0 {
			return ""
		}
		for _, c := range conds {
			cm, ok := c.(map[string]any)
			if !ok {
				continue
			}
			typ, _, _ := unstructured.NestedString(cm, "type")
			st, _, _ := unstructured.NestedString(cm, "status")
			if (typ == "Accepted" || typ == "ResolvedRefs" || typ == "Programmed") && st == "False" {
				reason, _, _ := unstructured.NestedString(cm, "reason")
				return "error:" + reason
			}
		}
		return "ok"
	case strings.HasSuffix(typeID, ".istio.io"):
		msgs, _, _ := unstructured.NestedSlice(u.Object, "status", "validationMessages")
		worst := ""
		for _, m := range msgs {
			mm, ok := m.(map[string]any)
			if !ok {
				continue
			}
			level, _, _ := unstructured.NestedString(mm, "level")
			code, _, _ := unstructured.NestedString(mm, "type", "code")
			if level == "ERROR" {
				return "error:" + code
			}
			if level == "WARNING" && worst == "" {
				worst = "warning:" + code
			}
		}
		return worst
	}
	return ""
}
```

```go
func newThrottle() *throttle { return &throttle{fails: map[string][]time.Time{}, now: time.Now} }

func (t *throttle) blocked(key string) time.Duration {
	t.mu.Lock()
	defer t.mu.Unlock()
	ts := t.trim(key)
	if len(ts) < throttleMax {
		return 0
	}
	if until := ts[len(ts)-1].Add(throttleBlock); t.now().Before(until) {
		return until.Sub(t.now())
	}
	return 0
}

func (t *throttle) fail(key string) {
	t.mu.Lock()
	defer t.mu.Unlock()
	t.fails[key] = append(t.trim(key), t.now())
	// ponytail: entries are trimmed on access only; a flood of distinct keys grows the map until each key's window passes.
}

func (t *throttle) reset(key string) { t.mu.Lock(); delete(t.fails, key); t.mu.Unlock() }

// trim drops failures older than the window; caller holds the lock.
func (t *throttle) trim(key string) []time.Time {
	cut := t.now().Add(-throttleWindow)
	ts := t.fails[key]
	i := 0
	for i < len(ts) && ts[i].Before(cut) {
		i++
	}
	ts = ts[i:]
	if len(ts) == 0 {
		delete(t.fails, key)
	} else {
		t.fails[key] = ts
	}
	return ts
}

func clientIP(r *http.Request) string {
	if xff := r.Header.Get("X-Forwarded-For"); xff != "" {
		return strings.TrimSpace(strings.Split(xff, ",")[0])
	}
	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err != nil {
		return r.RemoteAddr
	}
	return host
}
```
